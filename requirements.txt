# Runtime dependencies for the bot. Requires Python 3.14 on Windows.
#
#   py -3.14 -m venv venv
#   venv\Scripts\activate
#   pip install -r requirements.txt
#
# Versions are pinned to what the project is developed against. See
# docs/SETUP.md for the prerequisites pip cannot install (Tesseract-OCR,
# and a CUDA build of torch if you have an NVIDIA GPU).

# --- Windows integration -----------------------------------------------------
# The bot drives the game client through Win32 window handles and raw input;
# there is no cross-platform path.

keyboard==0.13.5

# --- Vision ------------------------------------------------------------------

dxcam==0.3.0
opencv-python==5.0.0.93
numpy==2.5.2
pillow>=13.0.0

# torch/torchvision below are the CPU wheels from PyPI. If you have an NVIDIA
# GPU, install the CUDA build INSTEAD -- see docs/SETUP.md, "GPU vs CPU".
torch==2.13.0
torchvision==0.28.0
ultralytics==8.4.135

# --- OCR (chat and UI text reading) ------------------------------------------
# Also needs the Tesseract-OCR binary installed separately; see docs/SETUP.md.
pytesseract==0.3.13

# --- Pathfinding and movement ------------------------------------------------
pathfinding==1.0.22
pytweening==1.2.0

# --- Config, profiles, notifications -----------------------------------------
PyYAML==6.0.3

# discord.py, NOT the `discord` shim package: that shim is unmaintained and
